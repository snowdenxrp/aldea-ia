package org.apache.kafka.metadata.authorizer;

import org.apache.kafka.common.Endpoint;
import org.apache.kafka.common.Uuid;
import org.apache.kafka.common.metrics.Metrics;
import org.apache.kafka.common.metrics.internals.PluginMetricsImpl;
import org.apache.kafka.common.resource.ResourcePattern;
import org.apache.kafka.common.security.auth.KafkaPrincipal;
import org.apache.kafka.common.security.auth.SecurityProtocol;
import org.apache.kafka.server.authorizer.Action;
import org.apache.kafka.server.authorizer.AuthorizationResult;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.Timeout;

import java.util.ArrayList;
import java.lang.reflect.Field;
import java.util.List;
import java.util.Map;
import java.util.concurrent.TimeUnit;

import static org.apache.kafka.common.acl.AclOperation.WRITE;
import static org.apache.kafka.common.acl.AclPermissionType.ALLOW;
import static org.apache.kafka.common.resource.PatternType.LITERAL;
import static org.apache.kafka.common.resource.ResourceType.TOPIC;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

@Timeout(value = 120)
public class NexoJmmCausalWindowTest {
    private static final Endpoint PLAINTEXT =
        new Endpoint("PLAINTEXT", SecurityProtocol.PLAINTEXT, "127.0.0.1", 9022);
    private static final String USER = "nexo-jmm-causal";
    private static final String TOPIC_NAME = "nexo-jmm-causal-topic";
    private static final int ITERATIONS = 100;
    private static final int READERS = 4;
    private static final long PRE_REMOVE_NANOS = TimeUnit.MILLISECONDS.toNanos(5);
    private static final long OBSERVATION_END_NANOS = TimeUnit.MILLISECONDS.toNanos(15);

    private static StandardAuthorizer newAuthorizer() {
        StandardAuthorizer authorizer = new StandardAuthorizer();
        authorizer.configure(Map.of(StandardAuthorizer.SUPER_USERS_CONFIG, "User:superman"));
        Metrics metrics = new Metrics();
        authorizer.withPluginMetrics(new PluginMetricsImpl(metrics, Map.of()));
        authorizer.start(new AuthorizerTestServerInfo(List.of(PLAINTEXT)));
        authorizer.completeInitialLoad();
        return authorizer;
    }

    private static StandardAcl acl() {
        return new StandardAcl(
            TOPIC, TOPIC_NAME, LITERAL, "User:" + USER, "*", WRITE, ALLOW);
    }

    private static org.apache.kafka.server.authorizer.AuthorizableRequestContext context() throws Exception {
        return new MockAuthorizableRequestContext.Builder()
            .setPrincipal(new KafkaPrincipal(KafkaPrincipal.USER_TYPE, USER))
            .build();
    }

    private static Action writeAction() {
        return new Action(
            WRITE,
            new ResourcePattern(TOPIC, TOPIC_NAME, LITERAL),
            0,
            false,
            false);
    }

    private static final class Observation {
        final long enter;
        final long exit;
        final AuthorizationResult result;
        final Object cacheBeforeAuthorize;

        Observation(long enter, long exit, AuthorizationResult result, Object cacheBeforeAuthorize) {
            this.enter = enter;
            this.exit = exit;
            this.result = result;
            this.cacheBeforeAuthorize = cacheBeforeAuthorize;
        }
    }

    private static Object aclCacheOf(StandardAuthorizer authorizer) throws Exception {
        Field dataField = StandardAuthorizer.class.getDeclaredField("data");
        dataField.setAccessible(true);
        Object data = dataField.get(authorizer);
        Field cacheField = data.getClass().getDeclaredField("aclCache");
        cacheField.setAccessible(true);
        return cacheField.get(data);
    }

    private static final class WriterObservation {
        long enter = Long.MAX_VALUE;
        long exit = Long.MAX_VALUE;
    }

    private static final class ClassificationCounts {
        long postReturnAllowed;
        long postReturnDenied;
        long overlapAllowed;
        long overlapDenied;
        long unexpected;
        long observations;
        long postReturnAllowedWithPreRemoveCache;
    }

    private static void classifyObservation(
        Observation observation,
        WriterObservation writerObservation,
        ClassificationCounts counts
    ) {
        counts.observations++;
        if (observation.result == null) {
            counts.unexpected++;
        } else if (observation.enter > writerObservation.exit) {
            classifyPostReturn(observation, counts);
        } else if (observation.enter <= writerObservation.exit
            && observation.exit >= writerObservation.enter) {
            classifyOverlap(observation, counts);
        }
    }

    private static void classifyPostReturn(
        Observation observation,
        ClassificationCounts counts
    ) {
        if (observation.result == AuthorizationResult.ALLOWED) {
            counts.postReturnAllowed++;
        } else if (observation.result == AuthorizationResult.DENIED) {
            counts.postReturnDenied++;
        } else {
            counts.unexpected++;
        }
    }

    private static void classifyOverlap(
        Observation observation,
        ClassificationCounts counts
    ) {
        if (observation.result == AuthorizationResult.ALLOWED) {
            counts.overlapAllowed++;
        } else if (observation.result == AuthorizationResult.DENIED) {
            counts.overlapDenied++;
        } else {
            counts.unexpected++;
        }
    }

    @Test
    public void causalPostReturnWindowIsDiagnosticOnly() throws Exception {
        ClassificationCounts counts = new ClassificationCounts();

        for (int iteration = 0; iteration < ITERATIONS; iteration++) {
            StandardAuthorizer authorizer = newAuthorizer();
            Uuid id = new Uuid(iteration + 501L, iteration + 1501L);
            authorizer.addAcl(id, acl());

            List<AuthorizationResult> baseline =
                authorizer.authorize(context(), List.of(writeAction()));
            assertEquals(AuthorizationResult.ALLOWED, baseline.get(0),
                "baseline authorization must be ALLOWED");

            long raceStart = System.nanoTime();
            long scheduledRemove = raceStart + PRE_REMOVE_NANOS;
            long observationEnd = raceStart + OBSERVATION_END_NANOS;
            WriterObservation writerObservation = new WriterObservation();
            @SuppressWarnings("unchecked")
            ArrayList<Observation>[] readerObservations = new ArrayList[READERS];
            for (int r = 0; r < READERS; r++) {
                readerObservations[r] = new ArrayList<>();
            }

            Thread writer = new Thread(() -> {
                while (System.nanoTime() < scheduledRemove) {
                    Thread.onSpinWait();
                }
                writerObservation.enter = System.nanoTime();
                authorizer.removeAcl(id);
                writerObservation.exit = System.nanoTime();
            }, "nexo-jmm-causal-writer");

            Thread[] readers = new Thread[READERS];
            for (int r = 0; r < READERS; r++) {
                final ArrayList<Observation> local = readerObservations[r];
                readers[r] = new Thread(() -> {
                    while (System.nanoTime() < observationEnd) {
                        long enter = System.nanoTime();
                        AuthorizationResult result;
                        Object cacheBeforeAuthorize;
                        try {
                            cacheBeforeAuthorize = aclCacheOf(authorizer);
                            result = authorizer.authorize(context(), List.of(writeAction())).get(0);
                        } catch (Throwable t) {
                            local.add(new Observation(enter, System.nanoTime(), null, null));
                            return;
                        }
                        long exit = System.nanoTime();
                        if (enter >= scheduledRemove) {
                            local.add(new Observation(enter, exit, result, cacheBeforeAuthorize));
                        }
                    }
                }, "nexo-jmm-causal-reader-" + r);
            }

            writer.start();
            for (Thread reader : readers) {
                reader.start();
            }

            writer.join(5000);
            for (Thread reader : readers) {
                reader.join(5000);
                assertTrue(!reader.isAlive(), "reader did not finish");
            }
            assertTrue(!writer.isAlive(), "writer did not finish");

            Object postRemoveCache = aclCacheOf(authorizer);
            for (ArrayList<Observation> local : readerObservations) {
                for (Observation observation : local) {
                    classifyObservation(observation, writerObservation, counts);
                    if (observation.result == AuthorizationResult.ALLOWED
                        && observation.enter > writerObservation.exit
                        && observation.cacheBeforeAuthorize == postRemoveCache) {
                        counts.postReturnAllowedWithPreRemoveCache++;
                    }
                }
            }
        }

        System.out.println(
            "CAUSAL_JMM_RACE ITERATIONS=" + ITERATIONS +
            " READERS=" + READERS +
            " OBSERVATIONS=" + counts.observations +
            " POST_RETURN_ALLOWED=" + counts.postReturnAllowed +
            " POST_RETURN_DENIED=" + counts.postReturnDenied +
            " OVERLAP_ALLOWED=" + counts.overlapAllowed +
            " OVERLAP_DENIED=" + counts.overlapDenied +
            " POST_RETURN_ALLOWED_WITH_PRE_REMOVE_CACHE=" + counts.postReturnAllowedWithPreRemoveCache +
            " UNEXPECTED=" + counts.unexpected);

        assertEquals(0, counts.unexpected, "diagnostic encountered unexpected execution failures");
    }

}
